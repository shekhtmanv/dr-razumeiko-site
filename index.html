/** * SPA Router & Navigation Controller */ // Функція переходу за внутрішніми посиланнями function navigate(url) { window.history.pushState({}, "", url); renderPage(url); } // Функція рендерингу сторінок залежно від шляху function renderPage(url) { const path = window.location.pathname; const searchParams = new URLSearchParams(window.location.search); // Приклад базового світча для різних розділів сайту const appContainer = document.getElementById('app'); if (!appContainer) return; if (path === '/' || path === '/index.html') { // Рендер головної сторінки // appContainer.innerHTML = ... } else if (path.startsWith('/blog')) { const postId = searchParams.get('id'); if (postId) { // Рендер конкретного поста блогу за ID console.log(`Loading blog post ID: ${postId}`); } else { // Рендер списку постів блогу console.log('Loading blog list'); } } // Додайте інші маршрути за потреби } // Обробка події "Назад/Вперед" у браузері window.addEventListener('popstate', () => { renderPage(window.location.pathname); }); // Головний глобальний обробник кліків document.addEventListener("click", e => { // 1. Обробка кліків по постах блогу const blogLink = e.target.closest("[data-blog-id]"); if (blogLink) { e.preventDefault(); const postId = blogLink.getAttribute("data-blog-id"); window.history.pushState({}, "", `/blog?id=${postId}`); renderPage('/blog'); return; } // 2. Обробка внутрішніх навігаційних посилань (SPA) const targetLink = e.target.closest("[data-link]"); if (targetLink) { e.preventDefault(); navigate(targetLink.getAttribute("href")); return; } // 3. Усі інші посилання (зовнішні соцмережі, Telegram, Instagram, телефони tel:) // ігноруються роутером і працюють за стандартною логікою браузера. }); // Ініціалізація при першому завантаженні сторінки document.addEventListener("DOMContentLoaded", () => { renderPage(window.location.pathname); });
